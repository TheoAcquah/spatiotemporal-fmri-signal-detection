
##############################################################
# ONE
##############################################################
docker run -ti --rm \
  -v ~/documents/Test_Retest_FMRI/bids_dataset:/data:ro \
  -v ~/documents/Test_Retest_FMRI/fmriprep_output:/out \
  -v ~/documents/license.txt:/opt/freesurfer/license.txt \
  -v ~/bids_filter.json:/bids_filter.json \
  nipreps/fmriprep:latest \
  /data /out participant \
  --participant-label sub-01 \
  --fs-license-file /opt/freesurfer/license.txt \
  --bids-filter-file /bids_filter.json \
  --output-spaces MNI152NLin2009cAsym:res-2 \
  --skip_bids_validation \
  --nthreads 8 --mem_mb 16000
